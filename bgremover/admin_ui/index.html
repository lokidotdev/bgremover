API usage

From a website on the allow-list

const form = new FormData();
form.append("image", fileInput.files[0]);
form.append("format", "png");          // png | webp | jpg | mask
// form.append("background", "#ffffff"); // optional solid background
// form.append("crop", "true");          // optional: trim to subject

const res = await fetch("/v1/remove", { method: "POST", body: form });
if (!res.ok) throw new Error((await res.json()).error);
const blob = await res.blob();           // the cutout

curl (needs "Allow requests without an Origin")

curl -F image=@photo.jpg "/v1/remove?format=png" -o cutout.png
